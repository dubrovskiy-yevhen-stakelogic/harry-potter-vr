cmake_minimum_required(VERSION 3.22.1)
project(hpvr_voice_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(HPVR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${HPVR_ROOT}/cmake/HPVRVoice.cmake")
hpvr_add_voice_decoder("${HPVR_ROOT}")
add_executable(hpvr_quest_voice_cast_tests "${HPVR_ROOT}/src/quest/tests/quest_voice_cast_tests.cpp")
target_include_directories(hpvr_quest_voice_cast_tests PRIVATE "${HPVR_ROOT}/src/quest/include")
add_executable(hpvr_quest_voice_decoder_tests "${HPVR_ROOT}/src/quest/tests/quest_voice_decoder_tests.cpp")
target_link_libraries(hpvr_quest_voice_decoder_tests PRIVATE hpvr_voice_decoder)
add_executable(hpvr_voice_keyword_probe voice_keyword_probe.cpp)
target_link_libraries(hpvr_voice_keyword_probe PRIVATE hpvr_voice_decoder)
if(ANDROID)
    add_library(hpvr_voice_android_compile_check STATIC
        "${HPVR_ROOT}/android/app/src/main/cpp/quest_voice_cast_android.cpp")
    target_link_libraries(hpvr_voice_android_compile_check PRIVATE hpvr_voice_decoder aaudio)
    target_compile_options(hpvr_voice_android_compile_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    add_executable(hpvr_quest_voice_recording_diagnostic_tests
        "${HPVR_ROOT}/src/quest/tests/quest_voice_recording_diagnostic_tests.cpp")
    target_include_directories(hpvr_quest_voice_recording_diagnostic_tests PRIVATE "${HPVR_ROOT}/src/quest/include")
    if(MSVC)
        target_compile_options(hpvr_quest_voice_recording_diagnostic_tests PRIVATE /W4 /WX)
    endif()
    add_executable(hpvr_quest_voice_worker_tests
        "${HPVR_ROOT}/src/quest/tests/quest_voice_worker_tests.cpp"
        "${HPVR_ROOT}/android/app/src/main/cpp/quest_voice_cast_android.cpp")
    target_include_directories(hpvr_quest_voice_worker_tests PRIVATE
        "${HPVR_ROOT}/src/quest/tests/voice_stubs"
        "${HPVR_ROOT}/src/quest/include"
        "${HPVR_ROOT}/android/app/src/main/cpp")
    if(MSVC)
        target_compile_options(hpvr_quest_voice_worker_tests PRIVATE /W4 /WX)
    endif()
    enable_testing()
    add_test(NAME voice_policy COMMAND hpvr_quest_voice_cast_tests)
    add_test(NAME voice_worker_lifecycle COMMAND hpvr_quest_voice_worker_tests)
    add_test(NAME voice_recording_diagnostic COMMAND hpvr_quest_voice_recording_diagnostic_tests)
    set(HPVR_VOICE_NEGATIVE_PCM "" CACHE FILEPATH "Optional local unrelated-speech PCM16 mono16k fixture")
    add_test(NAME voice_acoustic_negative COMMAND hpvr_quest_voice_decoder_tests
        "${HPVR_ROOT}/local/neural-voice-dependencies/assets/hpvr-voice"
        "${HPVR_VOICE_NEGATIVE_PCM}")
    foreach(voice_executable hpvr_quest_voice_decoder_tests hpvr_voice_keyword_probe)
        hpvr_copy_voice_runtime(${voice_executable})
    endforeach()
endif()
foreach(checked_target hpvr_voice_decoder hpvr_quest_voice_cast_tests hpvr_quest_voice_decoder_tests)
    if(MSVC)
        target_compile_options(${checked_target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${checked_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
